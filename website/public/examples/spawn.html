spawn

Green threads: spawn a worker that produces a value, receive it over a channel.

Run it with bit run examples/spawn.

// Green threads: spawn a worker that produces a value, receive it over a channel.
function producer(out: chan<i64>) {
  out <- 42
}

function main() {
  let out = chan<i64>(0)
  spawn producer(out)
  print("worker produced ${<- out}\n")
}
