Tuples: a multi-value return, destructuring it, and reading elements by index.

Run it with bit run examples/tuples.

// Tuples: a multi-value return, destructuring it, and reading elements by index.
// A tuple is the lightweight way to hand back more than one value without
// declaring a type for the pair - a struct is still the right answer when the
// members deserve names or need to change after the fact, because tuple elements
// are read-only.
function divmod(a: i64, b: i64): (i64, i64) {
  return a / b, a % b
}

// Element types are independent, and there is no arity limit.
function describe(n: i64): (string, i64, bool) {
  if (n < 0) {
    return "negative", 0 - n, false
  }
  return "positive", n, true
}

// Tuples nest: an element can itself be a tuple.
function split(a: i64, b: i64): ((i64, i64), i64) {
  return divmod(a, b), a * b
}

function main() {
  // Destructure straight out of the call.
  let (q, r) = divmod(17, 5)
  println("17 / 5 = ${q} remainder ${r}")

  // Or keep the tuple whole and index it. `t.0` is a read; tuple elements
  // cannot be assigned.
  let t = divmod(9, 4)
  println("9 / 4 = ${t.0} remainder ${t.1}")

  let (sign, magnitude, positive) = describe(0 - 42)
  println("-42 is ${sign}, magnitude ${magnitude}, positive=${positive}")

  // `_` skips an element you do not need.
  let (_, mag2, _) = describe(7)
  println("magnitude of 7 is ${mag2}")

  // A nested pattern unpacks both levels in one binding.
  let ((q2, r2), product) = split(23, 7)
  println("23 / 7 = ${q2} remainder ${r2}, product ${product}")

  // Destructuring can also assign to locals that already exist.
  let a = 0
  let b = 0
  (a, b) = divmod(100, 7)
  println("100 / 7 = ${a} remainder ${b}")
}
