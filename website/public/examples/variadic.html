Variadic parameters: `...name: T` binds one `[]T` inside the body (§10.3),

Run it with bit run examples/variadic.

// Variadic parameters: `...name: T` binds one `[]T` inside the body (§10.3),
// and a call may either list the trailing values or spread an existing slice
// with `...` (§12.4).

function total(...xs: i64): i64 {
  let t = 0
  for x of xs {
    t = t + x
  }
  return t
}

// Fixed parameters come first; only the last one may be variadic.
function labelled(label: string, ...xs: i64): string {
  return "${label}=${total(...xs)}"
}

function main() {
  print("${total()}\n")
  print("${total(1, 2, 3, 4)}\n")

  let nums = [10, 20, 30]
  print("${total(...nums)}\n")
  print("${labelled("sum", 5, 6)}\n")
}
