uuid

UUIDs (std/uuid, RFC 9562). v4 is fully random - the go-to for opaque unique

Run it with bit run examples/uuid.

// UUIDs (std/uuid, RFC 9562). v4 is fully random - the go-to for opaque unique
// ids. v7 packs a Unix-millisecond timestamp into its high bits, so v7 values
// sort in creation order, which makes them friendlier as database keys. Both
// carry random bits, so a fresh value differs every run; the printed lines are
// the fixed version/variant nibbles and a parse round-trip, so the output is
// deterministic.
import { uuidV4, uuidV7, parse, format, equals, nilUUID } from "std/uuid"

function main() {
  // v4: random. The version nibble is 4 and the variant is 2 (RFC 4122/9562).
  let a = uuidV4()
  println("v4ver=${a.version()}")
  println("v4var=${a.variant()}")

  // v7: time-ordered. The version nibble is 7.
  let b = uuidV7()
  println("v7ver=${b.version()}")

  // Format to the canonical 8-4-4-4-12 text, parse it back, and confirm the
  // recovered value is identical.
  let text = format(b)
  let back = parse(text) catch nilUUID()
  println("roundtrip=${equals(back, b)}")

  // Malformed input is rejected, not guessed.
  let bad = parse("not-a-uuid") catch nilUUID()
  println("rejectbad=${equals(bad, nilUUID())}")
}
