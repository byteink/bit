Higher-order slice helpers from std/seq: transform, keep, fold, and search.

Run it with bit run examples/seqops.

// Higher-order slice helpers from std/seq: transform, keep, fold, and search.
// `find` returns an Option<T>; unwrap it or supply a default with the prelude
// helpers (unwrap/unwrapOr, auto-imported). `mapped` is spelled with the extra
// letters because `map` is a reserved word (the map type).
import { mapped, filter, reduce, find } from "std/seq"

function main() {
  let nums = [1, 2, 3, 4, 5, 6]

  let squares = mapped<i64, i64>(nums, (n: i64) => n * n)
  let evens = filter<i64>(squares, (n: i64) => n % 2 == 0)
  let total = reduce<i64, i64>(evens, 0, (a: i64, n: i64) => a + n)
  println("even squares sum to ${total}")

  let firstBig = find<i64>(nums, (n: i64) => n > 4)
  println("first > 4: ${unwrapOr(firstBig, -1)}")
}
