// C-like enums with exhaustive `match` (§14.4, §16.4). A tiny traffic-light
// state machine: each state advances to the next and maps to a human label.
// Every `match` must cover every variant - add a fourth `Light` and the
// compiler flags each `match` that forgot it.
enum Light { Red, Green, Yellow }

function next(l: Light): Light {
  match (l) {
    Red => { return Light.Green }
    Green => { return Light.Yellow }
    Yellow => { return Light.Red }
  }
  return Light.Red
}

function label(l: Light): string {
  match (l) {
    Red => { return "stop" }
    Green => { return "go" }
    Yellow => { return "slow" }
  }
  return "?"
}

function main() {
  let l = Light.Red
  let i = 0
  while (i < 4) {
    println(label(l))
    l = next(l)
    i = i + 1
  }
}
