// An HTTP/2 server - which is just `serveTls`. There is no `serveH2`, and none is
// needed: `serveTls` advertises ALPN ["h2","http/1.1"] and serves each connection
// over whichever protocol the handshake selected, dispatching both to the same
// `(Request) => Response` handler. HTTP/2 is the ordinary outcome; HTTP/1.1 is the
// fallback for clients that cannot do better.
//
// Getting a 200 back over TLS would prove nothing on its own - an HTTP/1.1 exchange
// looks identical from the caller's side, which is exactly how a silent downgrade
// hides. Two things together rule one out here:
//   * the client offers "h2" as its ONLY ALPN protocol, so http/1.1 is not
//     something the server can legitimately select for this connection, and
//   * the protocol the handshake actually settled on is read back off a live
//     connection and *asserted* to be "h2".
// The assert is the real proof, and it is deliberately load-bearing: a regression
// that downgraded the exchange would fail this example rather than pass quietly.
// The offer alone would not be enough - on no ALPN overlap std/tls completes the
// handshake with no protocol selected (rather than sending no_application_protocol)
// and both ends then fall back to HTTP/1.1, so only the asserted value settles it.
//
// Handshake keys are random, so nothing key- or timing-derived is printed: the
// output is statuses, bodies, and the negotiated protocol, and the whole demo -
// server, client, and CA - runs offline on loopback.
import { serveTls, requestTls, ok, respond, Request, Response } from "std/http"
import { dial as tlsDial, newTlsConfig, newTrustStore, TlsConfig } from "std/tls"
import { decodeHex, pemEncode } from "std/crypto"
import { sleep, Millisecond } from "std/time"

// A fixed loopback port for the demo, distinct from every other example and test.
const port: int = 47161

function hx(s: string): []byte {
  return decodeHex(s) catch e {
    return []byte(0)
  }
}
function pemOf(label: string, h: string): string { return pemEncode(label, hx(h)) }

// One handler for every protocol: HTTP/2 streams and HTTP/1.1 requests both arrive
// here as a `Request`, so nothing in the routing knows which is in play.
function route(req: Request): Response {
  if (req.method == "POST") {
    return ok("echo:${req.body}")
  }
  if (req.path == "/") {
    return ok("hello over HTTP/2")
  }
  return respond(404, "not found")
}

// The server green thread. `serveTls` binds inside and loops forever; the process
// exits when `main` returns, so no explicit shutdown is needed.
function server() {
  serveTls("127.0.0.1", port, pemOf("CERTIFICATE", leafHex), pemOf("PRIVATE KEY", keyHex), route) catch e {
    println("server error: ${e.message()}")
  }
}

// The h2-only client config: trusts the test CA, pins the SNI to the leaf's SAN and
// the clock into the certificate's validity window, and - the point of the demo -
// offers "h2" as the sole ALPN protocol, leaving the server no way to downgrade.
function h2Cfg(): TlsConfig! {
  let cfg = newTlsConfig(newTrustStore(pemOf("CERTIFICATE", caHex))?)
  cfg.serverName = "localhost"
  cfg.alpn = ["h2"]
  cfg.nowUnix = 1800000000
  return cfg
}

// Issue one request over HTTP/2, retrying the dial with a short bounded sleep while
// the spawned server is still binding.
function fetch(method: string, path: string, body: string): Response! {
  let cfg = h2Cfg()?
  let attempt = 0
  while (attempt < 200) {
    let r = requestTls(method, "https://127.0.0.1:${port}${path}", body, cfg) catch e {
      sleep(10 * Millisecond)
      attempt = attempt + 1
      continue
    }
    return r
  }
  fail newError("server did not come up")
}

function run(): ()! {
  spawn server()

  // Request and response bodies both cross the HTTP/2 engine.
  let root = fetch("GET", "/", "")?
  println("GET / -> ${root.status} ${root.body}")
  let echo = fetch("POST", "/echo", "payload")?
  println("POST /echo -> ${echo.status} ${echo.body}")
  let missing = fetch("GET", "/nope", "")?
  println("GET /nope -> ${missing.status} ${missing.body}")

  // The proof. Read the protocol off a live connection and hold the demo to it -
  // "h2" is the only value consistent with those responses having been HTTP/2.
  let conn = tlsDial("127.0.0.1", port, h2Cfg()?)?
  let proto = conn.alpnProtocol()
  conn.close()
  assert(proto == "h2", "ALPN selected '${proto}', not h2 - this was not HTTP/2")
  println("negotiated ALPN -> ${proto}")
}

function main() {
  run() catch e {
    println("failed: ${e.message()}")
  }
}
