// String byte access (§12.2): `len(s)` is a string's byte count and `s[i]` is
// its `u8` byte at index `i` (out-of-range panics, like a slice). That's enough
// to write real string routines in Bit itself - here, counting a byte and a
// naive substring search, no builtins beyond `len`.

// How many times the byte `b` occurs in `s`.
function countByte(s: string, b: u8): i64 {
  let n = 0
  let i = 0
  while (i < len(s)) {
    if (s[i] == b) {
      n = n + 1
    }
    i = i + 1
  }
  return n
}

// Index of the first byte `b` in `s`, or -1 if absent.
function indexByte(s: string, b: u8): i64 {
  let i = 0
  while (i < len(s)) {
    if (s[i] == b) {
      return i
    }
    i = i + 1
  }
  return -1
}

function main() {
  let s = "mississippi"
  print("len=${len(s)}\n") // 11
  print("s_count=${countByte(s, 115)}\n") // 's' -> 4
  print("i_first=${indexByte(s, 105)}\n") // 'i' -> 1
  print("x_first=${indexByte(s, 120)}\n") // 'x' -> -1
}
