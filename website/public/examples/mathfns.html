// std/math: rounding and logarithms over f64 (each a hardware instruction or a
// libm-free implementation), plus integer helpers. Trigonometry is not here
// yet - see the module's note.
import { sqrt, pow, log2, log10, floor, ceil, round, clamp } from "std/math"
import { gcd, ipow, imax } from "std/math"

function main() {
  println("sqrt(2) = ${sqrt(2.0)}")
  println("2^10    = ${pow(2.0, 10.0)}")
  println("log2(1000)  = ${log2(1000.0)}")
  println("log10(1000) = ${log10(1000.0)}")

  println("floor(-2.5)=${floor(-2.5)} ceil(-2.5)=${ceil(-2.5)} round(-2.5)=${round(-2.5)}")
  println("clamp(9, 0, 5) = ${clamp(9.0, 0.0, 5.0)}")

  println("gcd(48, 18) = ${gcd(48, 18)}")
  println("3^5 = ${ipow(3, 5)}")
  println("max(3, 9) = ${imax(3, 9)}")
}
