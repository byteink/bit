strslice

String slicing `s[lo:hi]` (§12.6) and the `[]byte`↔`string` conversions

Run it with bit run examples/strslice.

// String slicing `s[lo:hi]` (§12.6) and the `[]byte`↔`string` conversions
// (§12.9). A string slice copies a byte range into a fresh string; `lo`
// defaults to 0 and `hi` to `len(s)`. Converting to `[]byte` and back
// round-trips the bytes - enough to write real text routines in Bit itself.

// Does `s` start with `prefix`? Compare the leading slice.
function startsWith(s: string, prefix: string): bool {
  if (len(prefix) > len(s)) {
    return false
  }
  return s[0:len(prefix)] == prefix
}

// Uppercase ASCII letters by editing bytes: string -> []byte -> edit -> string.
function shout(s: string): string {
  let b = []byte(s)
  let i = 0
  while (i < len(b)) {
    if (b[i] >= 97 && b[i] <= 122) { // 'a'..'z'
      b[i] = b[i] - 32
    }
    i = i + 1
  }
  return string(b)
}

function main() {
  let s = "hello world"
  println(s[0:5]) // hello
  println(s[6:]) // world

  let hasHello = startsWith(s, "hello")
  let hasWorld = startsWith(s, "world")
  println("${hasHello} ${hasWorld}") // true false

  println(shout(s)) // HELLO WORLD
}
