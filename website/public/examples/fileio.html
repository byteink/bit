std/fs, std/path and std/io: write a small tree, walk it, and read it back

Run it with bit run examples/fileio.

// std/fs, std/path and std/io: write a small tree, walk it, and read it back
// through a buffered reader. Fallible calls use `?` to propagate and `catch` to
// supply a default.
import { writeFile, readFile, appendFile, mkdir, remove, walk, exists, open } from "std/fs"
import { join, base, ext } from "std/path"
import { reader, stdout } from "std/io"

function readBuffered(path: string): string! {
  let f = open(path)?
  defer f.close()
  return reader(f.fd).readAll()
}

function build(root: string): ()! {
  mkdir(root)?
  writeFile(join(root, "notes.txt"), "first line\n")?
  appendFile(join(root, "notes.txt"), "second line\n")?
  mkdir(join(root, "data"))?
  writeFile(join(root, "data/values.csv"), "1,2,3\n")?
}

function main() {
  let root = "/tmp/bit-example-fileio"
  if (exists(root)) {
    println("leftover tree at ${root}; remove it first")
    return
  }

  build(root) catch e {
    println("setup failed: ${e.message()}")
    return
  }

  let w = stdout()
  for p of walk(root) catch []string(0) {
    w.writeLine("${base(p)} (${ext(p)})")
  }
  w.flush()

  println("notes:")
  print(readBuffered(join(root, "notes.txt")) catch "<unreadable>")

  // Tear the tree down, deepest paths first.
  for p of walk(root) catch []string(0) {
    remove(p) catch println("could not remove ${p}")
  }
  remove(join(root, "data")) catch println("could not remove data/")
  remove(root) catch println("could not remove ${root}")
  println("cleaned up: ${!exists(root)}")
}
