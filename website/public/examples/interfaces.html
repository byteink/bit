// Structural interfaces: any type with the right methods satisfies the
// interface - no `implements` needed - and calls dispatch dynamically. A
// heterogeneous []Shape holds Circles and Squares; each area() call resolves to
// the concrete type's method at runtime.
interface Shape { area(): f64 }

struct Circle { export r: f64 }
function (c: Circle) area(): f64 { return 3.14159 * c.r * c.r }

struct Square { export s: f64 }
function (sq: Square) area(): f64 { return sq.s * sq.s }

function totalArea(shapes: []Shape): f64 {
  let sum = 0.0
  for s of shapes {
    sum = sum + s.area()
  }
  return sum
}

// A type assertion narrows an interface value back to a concrete type. The
// two-result form reports a mismatch as `ok = false` instead of panicking, which
// is what lets one loop ask "is this one a Circle?" of a mixed list.
function countCircles(shapes: []Shape): int {
  let n = 0
  for s of shapes {
    let (c, ok) = s.(Circle)
    if (ok) {
      n = n + 1
    }
  }
  return n
}

function main() {
  let shapes: []Shape = [Circle{ r: 1.0 }, Square{ s: 2.0 }, Circle{ r: 3.0 }]
  print("total area = ${totalArea(shapes)}\n")
  print("circles = ${countCircles(shapes)}\n")
}
