hashing

SHA-256 and HMAC-SHA256 over a message (std/crypto). A hash fingerprints

Run it with bit run examples/hashing.

// SHA-256 and HMAC-SHA256 over a message (std/crypto). A hash fingerprints
// data: the same input always yields the same 32-byte digest, and any change
// avalanches into a completely different one. An HMAC binds that fingerprint to
// a secret key, so only a holder of the key could have produced the tag - the
// basis of message authentication. Both are printed as lowercase hex, so the
// output is deterministic and independently checkable.
import { Hash, newSha256, digest, hmac, encodeHex } from "std/crypto"

// `hmac` is generic over the digest and takes a `() => Hash` value. `newSha256`
// returns the concrete `Sha256`, so wrap it to widen to the `Hash` interface.
function sha256Hash(): Hash { return newSha256() }

function main() {
  let msg = []byte("the quick brown fox")

  // A plain SHA-256 digest of the message.
  println("sha256=${encodeHex(digest(newSha256(), msg))}")

  // HMAC-SHA256: the same digest, keyed with a shared secret.
  let key = []byte("shared-secret")
  println("hmac=${encodeHex(hmac(sha256Hash, key, msg))}")
}
