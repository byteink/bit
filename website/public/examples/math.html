// std/math in use: Euclidean distance (needs sqrt) plus a clamp. The module is
// written in Bit itself, on top of numeric conversions and the one hardware
// primitive, sqrt.

import { pi, sqrt, abs, clamp } from "std/math"

// Straight-line distance between two points.
function distance(x1: f64, y1: f64, x2: f64, y2: f64): f64 {
  let dx = x2 - x1
  let dy = y2 - y1
  return sqrt(dx * dx + dy * dy)
}

function main() {
  let d = distance(0.0, 0.0, 3.0, 4.0)
  println("distance = ${d}")

  // The `pi` constant is an imported `export const`, inlined at use.
  let r = 2.0
  println("area = ${pi * r * r}")

  // Keep a value within a range.
  let v = clamp(abs(-42.0), 0.0, 10.0)
  println("clamped = ${v}")
}
