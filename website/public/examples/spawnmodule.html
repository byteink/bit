spawnmodule

`spawn` and `defer` across a module boundary and over generics.

Run it with bit run examples/spawnmodule.

spawnmodule.bit

// `spawn` and `defer` across a module boundary and over generics.
//
// Every callee here resolves through the same path a direct call uses, so the
// symbol emitted at the spawn site is the symbol the instance is defined under.
// Before the fix each of these built cleanly and died at dyld with an
// unresolved `_work` / `_relay` / `_tally`.
import { work, relay } from "./worker"

// Same-module generic. `T` binds to a SCALAR here: the thunk slot holding `v`
// must NOT appear in the trampoline's `ptr_offsets`. A layout built from the
// unsubstituted template types marks it a GC reference, and the collector then
// walks the integer 7 as a pointer.
function emit<T>(v: T, out: chan<T>) {
  out <- v
}

function tally<T>(label: string, n: T) {
  println("deferred ${label}=${n}")
}

function main() {
  // Deferred generic: runs at function exit, so it prints last.
  defer tally("total", 6)

  let ints = chan<int>(1)
  spawn work(5, ints)
  println("imported = ${<- ints}")

  spawn relay(11, ints)
  println("imported generic = ${<- ints}")

  spawn emit(7, ints)
  println("local generic = ${<- ints}")

  // A REFERENCE-typed instantiation of the same template, so both an int and a
  // pointer slot are exercised through one generic's thunk layout.
  let strs = chan<string>(1)
  spawn emit("hello", strs)
  println("ref generic = ${<- strs}")

  // A loop so `bit_rt_safepoint` fires (codegen emits one only at a loop
  // back-edge): without it a straight-line program collects zero times and a
  // BIT_GC=stress run proves nothing about the thunk's ptr_offsets.
  let sum = 0
  let i = 0
  while (i < 64) {
    spawn emit(i, ints)
    sum = sum + <- ints
    i = i + 1
  }
  println("loop sum = ${sum}")
}
