channels

A buffered channel: send three values, then receive them back. Then `close`

Run it with bit run examples/channels.

// A buffered channel: send three values, then receive them back. Then `close`
// it and drain with the two-result receive `let (v, ok) = <- c`: receives take
// whatever is still buffered, and `ok` goes false once the channel is closed and
// empty - which is how a receiver knows to stop.
function main() {
  let nums = chan<i64>(3)
  nums <- 1
  nums <- 2
  nums <- 3
  let a = <- nums
  let b = <- nums
  let c = <- nums
  print("received ${a} ${b} ${c}\n")

  let more = chan<i64>(2)
  more <- 10
  more <- 20
  close(more)

  let sum = 0
  let open = true
  while (open) {
    let (v, ok) = <- more
    if (ok) {
      sum = sum + v
    } else {
      open = false
    }
  }
  print("drained sum=${sum}\n")
}
