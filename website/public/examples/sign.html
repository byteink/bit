sign

An if / else-if / else chain classifying a number's sign.

Run it with bit run examples/sign.

// An if / else-if / else chain classifying a number's sign.
function sign(n: i64) {
  if (n > 0) {
    print("positive\n")
  } else {
    if (n < 0) {
      print("negative\n")
    } else {
      print("zero\n")
    }
  }
}

function main() {
  sign(7)
  sign(0)
  sign(-3)
}
