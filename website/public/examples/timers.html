std/time: durations are plain int nanoseconds, and `sleep` parks the green

Run it with bit run examples/timers.

// std/time: durations are plain int nanoseconds, and `sleep` parks the green
// thread instead of blocking its OS thread - so N concurrent sleepers finish in
// about one sleep, not N of them.
import { sleep, monotonic, since, toMillis, Millisecond } from "std/time"

function worker(id: i64, done: chan<i64>) {
  sleep(30 * Millisecond)
  done <- id
}

function main() {
  let done = chan<i64>(8)
  let start = monotonic()

  let i = 0
  while (i < 8) {
    spawn worker(i, done)
    i = i + 1
  }

  let sum = 0
  i = 0
  while (i < 8) {
    sum = sum + <- done
    i = i + 1
  }

  let elapsed = since(start)
  println("8 workers finished, id sum = ${sum}")
  println("each slept 30ms; overlapped: ${toMillis(elapsed) < 200.0}")
}
