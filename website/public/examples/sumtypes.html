Sum types (§14.7): a variant can carry data, and a payload-binding `match`

Run it with bit run examples/sumtypes.

// Sum types (§14.7): a variant can carry data, and a payload-binding `match`
// unpacks it (§13.8). A checked division returns either the quotient or an
// error - the classic Result shape, spelled out by hand. `Ok` carries an i64;
// `DivByZero` carries nothing; the `match` arm `Ok(q) =>` binds the payload.

enum DivResult { Ok(i64), DivByZero }

function checkedDiv(a: i64, b: i64): DivResult {
  if (b == 0) {
    return DivResult.DivByZero
  }
  return DivResult.Ok(a / b)
}

function report(a: i64, b: i64) {
  match (checkedDiv(a, b)) {
    Ok(q) => println("${a}/${b} = ${q}")
    DivByZero => println("${a}/${b} = error (divide by zero)")
  }
}

function main() {
  report(10, 2)
  report(7, 0)
  report(20, 4)
}
