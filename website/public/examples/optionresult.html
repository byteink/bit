Option<T> and Result<T,E> are generic enums (§14.7) that live in the prelude

Run it with bit run examples/optionresult.

// Option<T> and Result<T,E> are generic enums (§14.7) that live in the prelude
// (std/core), so they need no import. One declaration is monomorphized per use:
// Option<i64> here, Result<i64,string> below. Type arguments are inferred from
// the function's return type (or from Some's argument) - you never spell them at
// the construction site. A `match` unpacks the payload and forces the empty /
// error case to be handled, the type-safe alternative to a null or an error code.

// The first element that satisfies `> threshold`, or None if there is none.
function firstAbove(xs: []i64, threshold: i64): Option<i64> {
  for x of xs {
    if (x > threshold) {
      return Option.Some(x)
    }
  }
  return Option.None
}

// Integer division that reports the error case instead of trapping on b == 0.
function safeDiv(a: i64, b: i64): Result<i64, string> {
  if (b == 0) {
    return Result.Err("division by zero")
  }
  return Result.Ok(a / b)
}

function main() {
  let xs = [4, 15, 8, 23, 16]

  match (firstAbove(xs, 20)) {
    Some(v) => println("first above 20: ${v}")
    None => println("none above 20")
  }
  match (firstAbove(xs, 99)) {
    Some(v) => println("first above 99: ${v}")
    None => println("none above 99")
  }

  match (safeDiv(84, 2)) {
    Ok(q) => println("84 / 2 = ${q}")
    Err(m) => println("error: ${m}")
  }
  match (safeDiv(1, 0)) {
    Ok(q) => println("1 / 0 = ${q}")
    Err(m) => println("error: ${m}")
  }
}
