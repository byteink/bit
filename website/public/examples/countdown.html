countdown

A countdown printed as a shrinking row of stars.

Run it with bit run examples/countdown.

// A countdown printed as a shrinking row of stars.
function main() {
  let n = 5
  while (n > 0) {
    print("*")
    n = n - 1
  }
  print("\n")
}
