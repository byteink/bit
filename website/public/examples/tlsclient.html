// A TLS 1.3 client (std/tls). This builds a `TlsConfig`, then shows the
// `dial(host, 443, cfg)` HTTPS flow: open a TLS connection, write a bare HTTP
// request, read the first bytes of the reply, and close. The live dial is gated
// behind BIT_TLS_LIVE=1 so the demo is deterministic and non-blocking offline -
// CI prints a fixed "live: skipped" line and never touches the network. Set the
// variable to exercise a real handshake against a public server.
import { TlsConfig, newTlsConfig, emptyTrustStore, dial } from "std/tls"
import { env } from "std/os"

// A client config for a public HTTPS host. `verify` is off only because this
// seed build ships no bundled root store yet; a verifying client checks the
// chain against a trust store (see the httpsserver example, which does verify).
function clientConfig(host: string): TlsConfig {
  let cfg = newTlsConfig(emptyTrustStore())
  cfg.verify = false
  cfg.serverName = host
  cfg.alpn = ["http/1.1"]
  return cfg
}

// Dial `host:443`, send a minimal GET, and return the reply's first bytes.
function probe(host: string): string! {
  let conn = dial(host, 443, clientConfig(host))?
  conn.write([]byte("GET / HTTP/1.1\r\nHost: ${host}\r\nConnection: close\r\n\r\n"))?
  let head = conn.read(16)?
  conn.close()
  return string(head)
}

function main() {
  let host = "example.org"
  let cfg = clientConfig(host)
  println("serverName=${cfg.serverName}")
  println("alpn=${cfg.alpn[0]}")
  println("verify=${cfg.verify}")

  // Gate the network call so offline CI stays deterministic and never blocks.
  if (env("BIT_TLS_LIVE") != "1") {
    println("live: skipped")
    return
  }
  let line = probe(host) catch e {
    println("live: offline")
    return
  }
  println("live: ${line}")
}
