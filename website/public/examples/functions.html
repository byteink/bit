// Functions with parameters and a return value.
function max(a: i64, b: i64): i64 {
  if (a > b) {
    return a
  }
  return b
}

function main() {
  let bigger = max(3, 8)
  if (bigger == 8) {
    print("8 is bigger\n")
  }
}
