A 3x3 grid drawn with nested loops.

// A 3x3 grid drawn with nested loops.
function main() {
  for (let y = 0; y < 3; y = y + 1) {
    for (let x = 0; x < 3; x = x + 1) {
      print("#")
    }
    print("\n")
  }
}
