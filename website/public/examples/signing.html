Ed25519 digital signatures (std/crypto). A key pair is a 32-byte secret seed

Run it with bit run examples/signing.

// Ed25519 digital signatures (std/crypto). A key pair is a 32-byte secret seed
// (the private key) and the public key derived from it. The holder of the seed
// signs a message; anyone with the public key can verify the signature, but
// cannot forge one or recover the seed. The seed is drawn fresh from the OS
// CSPRNG, so the keys and signature differ every run; the printed lines are
// lengths and booleans, so the output is deterministic.
import { ed25519PublicKey, ed25519Sign, ed25519Verify, randomBytes } from "std/crypto"

function main() {
  // Generate a key pair: the 32-byte seed is the private key.
  let seed = randomBytes(32)
  let pub = ed25519PublicKey(seed)
  println("pubLen=${len(pub)}")

  let msg = []byte("transfer 100 to alice")
  let sig = ed25519Sign(seed, msg)
  println("sigLen=${len(sig)}")

  // The genuine signature verifies under the public key.
  println("verify=${ed25519Verify(pub, msg, sig)}")

  // A signature is bound to the exact message bytes: a tampered message fails.
  println("tamperVerify=${ed25519Verify(pub, []byte("transfer 900 to alice"), sig)}")
}
