// `match` as an expression (§13.8): the whole `match` yields a value, so it can
// be a `return`, a `let` initializer, or an operand - no temporary variable and
// no fallthrough. Each arm is exhaustive over the enum, and the arms must agree
// on a common type. Compare with examples/enums, which uses the statement form.
enum Suit { Clubs, Diamonds, Hearts, Spades }

// A `match` expression as the return value.
function color(s: Suit): string {
  return match (s) {
    Clubs => "black"
    Spades => "black"
    Diamonds => "red"
    Hearts => "red"
  }
}

// A `match` expression bound to a `let`, then used in arithmetic.
function points(s: Suit): i64 {
  let base = match (s) {
    Clubs => 1
    Diamonds => 2
    Hearts => 3
    Spades => 4
  }
  return base * 10
}

function main() {
  let hand = [Suit.Hearts, Suit.Spades, Suit.Diamonds]
  for s of hand {
    // A `match` expression embedded directly in a string interpolation.
    println("${color(s)} worth ${points(s)}")
  }
}
