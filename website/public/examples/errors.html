Error handling (§18): fallible functions return "a value OR an error".

Run it with bit run examples/errors.

// Error handling (§18): fallible functions return "a value OR an error".
//
//   T!          a result carrying an ok value of T or an error
//   fail e      produce the error (terminates like `return`)
//   expr?       propagate: unwrap on ok, return the error to the caller on fail
//   expr catch  handle locally - with a default value, or a binding block
//
// `newError(msg)` (from the core prelude) builds an error whose `message()`
// satisfies the predeclared `error` interface. Control flow stays visible:
// every early exit is a `?` or `fail` you can see, never a hidden throw.

function half(n: i64): i64! {
  if (n % 2 != 0) {
    fail newError("not even: ${n}")
  }
  return n / 2
}

// `?` chains: any odd input short-circuits out with its error.
function quarter(n: i64): i64! {
  let h = half(n)?
  return half(h)?
}

function main() {
  // ok path - `catch` supplies a fallback that is never used here
  print("quarter(20) = ${quarter(20) catch -1}\n")

  // error path - the fallback is taken
  print("quarter(7)  = ${quarter(7) catch -1}\n")

  // full handling - inspect the error through its interface method
  let v = quarter(6) catch e {
    print("handled: ${e.message()}\n")
    0
  }
  print("quarter(6)  = ${v}\n")
}
