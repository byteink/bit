syncmutex

std/sync KAT: N spawned tasks increment a shared counter under a Mutex.

Run it with bit run examples/syncmutex.

// std/sync KAT: N spawned tasks increment a shared counter under a Mutex.
//
// Runs at the scheduler's default worker count (one OS worker per CPU,
// )- genuinely concurrent, not the old single-worker serialization -
// so this is the actual proof that Mutex.lock()/unlock() correctly excludes
// concurrent green threads across real OS threads, not just cooperative
// ones on one. WaitGroup coordinates the fan-out/fan-in. Wrong under any
// data race, so a broken Mutex turns this red, not just slow.
import { Mutex, WaitGroup, newMutex, newWaitGroup } from "std/sync"

const workers = 32
const itersPerWorker = 5000

function main() {
  let counter = []i64(1)
  let mu = newMutex()
  let wg = newWaitGroup()

  wg.add(workers)
  let w = 0
  while (w < workers) {
    spawn bump(counter, mu, wg)
    w = w + 1
  }
  wg.wait()

  let want = workers * itersPerWorker
  if (counter[0] != want) {
    panic("counter: got ${counter[0]}, want ${want}")
  }
  print("counter: ${counter[0]} (exact - ${workers} workers x ${itersPerWorker} each)\n")
}

function bump(counter: []i64, mu: Mutex, wg: WaitGroup) {
  let i = 0
  while (i < itersPerWorker) {
    mu.lock()
    counter[0] = counter[0] + 1
    mu.unlock()
    i = i + 1
  }
  wg.done()
}
