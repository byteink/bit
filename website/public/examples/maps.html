Run it with bit run examples/maps.

// The map<K,V> type: literals, index get/set, the zero value for an absent key,
// the two-result index that reports presence, delete, len, and (key, value)
// iteration. Keys may be strings (byte-hashed) or any comparable scalar
// (word-hashed); an absent key reads as V's zero value.
function main() {
  // A word-count over a fixed list, built by incrementing map entries - the
  // absent-key-reads-zero rule makes `counts[w] + 1` work on first sight.
  let words = ["bit", "go", "bit", "rust", "go", "bit"]
  let counts = map<string, int>()
  for w of words {
    counts[w] = counts[w] + 1
  }
  println("bit=${counts["bit"]} go=${counts["go"]} rust=${counts["rust"]} zig=${counts["zig"]}")
  println("distinct=${len(counts)}")

  // A map literal with int keys, then delete and a presence-by-len check.
  let squares = map<int, int>{ 1: 1, 2: 4, 3: 9, 4: 16 }
  delete(squares, 3)
  let total = 0
  for (k, v) of squares {
    total = total + v
  }
  println("kept=${len(squares)} sumOfSquares=${total} three=${squares[3]}")

  // `squares[3]` reads 0 now that it is deleted - but so would a key whose value
  // really is 0. The two-result index is the only way to tell them apart.
  let stock = map<string, int>{ "apples": 0 }
  let (apples, hasApples) = stock["apples"]
  let (pears, hasPears) = stock["pears"]
  println("apples=${apples} present=${hasApples} / pears=${pears} present=${hasPears}")
}
