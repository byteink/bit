AES-256-GCM authenticated encryption (std/crypto). GCM is an AEAD: one call

Run it with bit run examples/aead.

// AES-256-GCM authenticated encryption (std/crypto). GCM is an AEAD: one call
// both encrypts the message and authenticates it together with associated data
// (AAD) that travels in the clear - a protocol header, say. `open` recovers the
// plaintext only if the key, nonce, ciphertext, and AAD are all intact;
// tampering with any of them fails closed instead of leaking a forged message.
//
// The key and nonce are drawn fresh from the OS CSPRNG, so the ciphertext
// differs every run; the printed lines are lengths and booleans, so the output
// stays deterministic.
import { newGcm, randomBytes, encodeHex } from "std/crypto"

function main() {
  let key = randomBytes(32) // 32 bytes selects AES-256
  let nonce = randomBytes(12) // GCM's standard 96-bit nonce
  let plaintext = []byte("attack at dawn")
  let aad = []byte("v1;msg-42") // authenticated, but sent in the clear

  let g = newGcm(key) catch e {
    println("keying failed: ${e.message()}")
    return
  }

  // seal -> ciphertext followed by a 16-byte authentication tag.
  let sealed = g.seal(nonce, plaintext, aad)
  println("sealedLen=${len(sealed)}")

  // open with the matching key/nonce/AAD recovers the message exactly.
  let opened = g.open(nonce, sealed, aad) catch e {
    println("open failed: ${e.message()}")
    return
  }
  println("roundtrip=${encodeHex(opened) == encodeHex(plaintext)}")

  // Changing the AAD (or the ciphertext) makes open reject - no plaintext leaks.
  let forged = g.open(nonce, sealed, []byte("v1;msg-43")) catch e {
    println("tamperRejected=true")
    return
  }
  println("tamperRejected=false")
}
