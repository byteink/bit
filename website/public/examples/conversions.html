// Numeric conversions T(x) (§12.9). Bit has NO implicit numeric conversions -
// not even i32→i64 - so mixing widths, or ints with floats, needs an explicit
// cast. Here: sum a string's bytes as integers, then average them as a float.

function main() {
  let s = "Hello"

  // Each s[i] is a u8; widen to i64 to accumulate without overflow.
  let sum = 0
  let i = 0
  while (i < len(s)) {
    sum = sum + i64(s[i])
    i = i + 1
  }

  // Average as a float: convert both operands so the division is real, not
  // integer division.
  let avg = f64(sum) / f64(len(s))
  println("sum=${sum} avg=${avg}")

  // And back: truncate the float average to an integer.
  println("truncated=${i64(avg)}")
}
