// The std/strings module: byte-level search, ASCII case-folding, join/split,
// int parsing, and a concat Builder (whose methods return an updated value, so
// reassign to chain).
import { hasPrefix, toUpper, join, indexOf, split, trimSpace, parseInt, newBuilder } from "std/strings"

// Sum a comma-separated list of integers, tolerating surrounding spaces.
function sumCsv(line: string): int! {
  let total = 0
  for field of split(line, ",") {
    total = total + parseInt(trimSpace(field))?
  }
  return total
}
