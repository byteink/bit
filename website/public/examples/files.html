File I/O with std/fs - the payoff of error handling + defer working together.

Run it with bit run examples/files.

// File I/O with std/fs - the payoff of error handling + defer working together.
//
// open/create return a File or an error, so `defer f.close()` guarantees the
// handle is released on every exit path. readFile/writeFile wrap the whole
// open→use→close dance. Errors are values: `?` propagates, `catch` handles.

import { open, readFile, writeFile } from "std/fs"

function main(): ()! {
  let path = "/tmp/bit_example_files.txt"

  // Write, then read the whole thing back. `?` propagates any error to main,
  // which the runtime reports on stderr with a non-zero exit (§17.4).
  writeFile(path, "line one\nline two\nline three\n")?
  let text = readFile(path)?
  print(text)

  // Open a handle explicitly to show the defer-close pattern; count the lines.
  let f = open(path)?
  defer f.close()
  let body = f.readAll()
  let lines = 0
  let i = 0
  while (i < len(body)) {
    if (body[i] == 10) { // '\n'
      lines = lines + 1
    }
    i = i + 1
  }
  println("lines=${lines}")
}
