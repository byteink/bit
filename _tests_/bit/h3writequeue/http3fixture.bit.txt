import { H3QueueFixture, h3WriteQueueFixture } from "std/quic"
import { Timer, Second, Millisecond } from "std/time"

interface h3wqCleanupFailure { cause: error, message(): string }

fn h3wqCall(t: H3Stream, wake: chan<int>, done: chan<error>) {
  t.writeUntil("DATA", wake) catch e {
    done <- e
    return
  }
  done <- newError("unexpected write success")
}

fn h3wqDeadline(wake: chan<int>, done: chan<int>) {
  let timer = Timer(10 * Millisecond)
  defer {
    let _ = timer.stop()
  }
  let _ = <- timer.c
  wake <- 1
  done <- 1
}

fn h3wqWait(f: H3QueueFixture, done: chan<error>): error! {
  let timer = Timer(Second)
  defer {
    let _ = timer.stop()
  }
  select {
    case e = <- done:
      return e
    case _ = <- timer.c:
      f.close()
      let _ = <- done
      fail newError("H3 write ignored cancellation bound during live full64 cleanup")
  }
}

fn h3wqRun(prequeued: bool): ()! {
  let f = h3WriteQueueFixture()
  defer f.close()
  let t = H3Stream{ reader = h3TunnelReader{ s = f.stream } }
  let wake = chan<int>(1)
  let sent = chan<int>(1)
  if (prequeued) {
    wake <- 1
    sent <- 1
  } else {
    spawn h3wqDeadline(wake, sent)
  }
  let done = chan<error>(1)
  spawn h3wqCall(t, wake, done)
  let e = h3wqWait(f, done)?
  let _ = <- sent
  let (incomplete, ok) = e.(h3wqCleanupFailure)
  assert(ok && t.closed, "incomplete cleanup preserves a terminal local writer")
  let (_, aborted) = incomplete.cause.(H3Aborted)
  assert(aborted, "original cancellation remains typed H3Aborted")
  assert(f.live(), "bounded cleanup must not abort the connection")
  assert(f.preserved(), "all64 prior FIFO commands remain unchanged")
  t.write("again") catch closed {
    assert(closed.message() == "h3: tunnel is closed", closed.message())
    return
  }
  assert(false, "terminal writer must reject another DATA frame")
}

test "actual H3 write cancellation returns before live full64 cleanup" {
  h3wqRun(true) catch e {
    assert(false, e.message())
  }
}

test "actual H3 write deadline wake returns before live full64 cleanup" {
  h3wqRun(false) catch e {
    assert(false, e.message())
  }
}
