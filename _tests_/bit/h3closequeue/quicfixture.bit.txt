// Added only to the disposable stdlib overlay; never a production export.
export class H3QueueFixture {
  export stream: Stream
  closed: bool

  export live(): bool {
    select {
      case _ = <- this.stream.doneCh:
        return false
      default:
        return true
    }
  }

  export close() {
    if (!this.closed) {
      this.closed = true
      close(this.stream.doneCh)
    }
  }

  export preserved(): bool {
    let count = 0
    while (count < 64) {
      select {
        case cmd = <- this.stream.cmdCh:
          if (cmd.kind != quicconnCmdWriteDrained || cmd.id != count) {
            return false
          }
        default:
          return false
      }
      count = count + 1
    }
    return true
  }
}

export fn h3CloseQueueFixture(): H3QueueFixture {
  let s = Stream{
    streamId = 0, cmdCh = chan<quicconnCmd>(64), recvCh = chan<StreamChunk>(1),
    doneCh = chan<int>(1), tx = quicconnNewReadTx(chan<int>(1)),
    acct = quicconnReadAcct{},
  }
  let i = 0
  while (i < 64) {
    s.cmdCh <- quicconnCmd{
      kind = quicconnCmdWriteDrained, id = i, data = []byte(0),
      replyOpen = chan<quicconnOpenReply>(1), replyInt = chan<int>(1),
    }
    i = i + 1
  }
  return H3QueueFixture{ stream = s }
}
