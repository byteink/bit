import { H3QueueFixture, h3CloseQueueFixture } from "std/quic"
import { Timer, Second, Millisecond, monotonic } from "std/time"

fn h3cqCall(t: H3Stream, bound: H3Bound, done: chan<error>) {
  t.closeUntil(bound) catch e {
    done <- e
    return
  }
  done <- newError("unexpected close success")
}

fn h3cqWait(f: H3QueueFixture, done: chan<error>): error! {
  let timer = Timer(Second)
  defer {
    let _ = timer.stop()
  }
  select {
    case e = <- done:
      return e
    case _ = <- timer.c:
      f.close()
      let _ = <- done
      fail newError("H3 close exceeded its bound before live full64 cleanup")
  }
}

fn h3cqRun(woken: bool): ()! {
  let f = h3CloseQueueFixture()
  defer f.close()
  let t = H3Stream{ reader = h3TunnelReader{ s = f.stream } }
  let bound = H3Bound{ wake = chan<int>(1), deadlineNs = monotonic().ns + 10 * Millisecond }
  if (woken) {
    bound.wake <- 1
  }
  let done = chan<error>(1)
  spawn h3cqCall(t, bound, done)
  let e = h3cqWait(f, done)?
  let (incomplete, ok) = e.(H3CleanupIncomplete)
  assert(ok && t.closed, "actual H3 close reports incomplete cleanup and stays terminal")
  assert(f.live(), "cleanup did not close the connection actor")
  assert(f.preserved(), "all64 prior FIFO commands remain unchanged")
  if (woken) {
    let (_, aborted) = incomplete.cause.(H3Aborted)
    assert(aborted, "wake cause remains typed H3Aborted")
  } else {
    assert(incomplete.cause.message() == "quic: stream deadline exceeded", e.message())
  }
  t.closeUntil(bound)?
  t.write("again") catch closed {
    assert(closed.message() == "h3: tunnel is closed", closed.message())
    return
  }
  assert(false, "terminal stream cannot accept another write")
}

test "actual H3 close deadline returns before live full64 cleanup" {
  h3cqRun(false) catch e {
    assert(false, e.message())
  }
}

test "actual H3 close wake returns before live full64 cleanup" {
  h3cqRun(true) catch e {
    assert(false, e.message())
  }
}
